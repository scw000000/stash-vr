# /browse page — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `/browse` HTML surface served by stash-vr, viewed inside DeoVR's built-in browser, that lets the user browse scenes by performer/studio/tag, view scene details, and rate/favorite/tag-edit/o-counter/organized-toggle directly — all without bouncing through HereSphere or DeoVR's player API.

**Architecture:** Server-rendered Go templates (same pattern as the existing `/` config page). One persistent right-side sidebar with three entity tabs (performers/studios/tags) plus a scene grid that varies by URL. A scene detail page provides write-back to Stash via `library.Service` methods that already exist for the HereSphere event handler. No SPA, no HTMX, only progressive-enhancement vanilla JS for the sidebar filter input. The `[Play in DeoVR]` button hands off to the existing `/deovr/videoData/{id}` URL.

**Tech Stack:** Go 1.24, `github.com/go-chi/chi/v5`, `html/template`, embedded `embed.FS` static, `github.com/Khan/genqlient` for Stash GraphQL, `golang.org/x/sync/singleflight` for caching, `zerolog`. No new dependencies.

**Reference spec:** [docs/superpowers/specs/2026-05-07-browse-page-design.md](../specs/2026-05-07-browse-page-design.md)

**Verification convention:** the project has no test suite (per CLAUDE.md). Each task's verification is `go vet ./...`, `go build ./...`, and (where relevant) a manual smoke test against a running server pointed at a Stash instance.

- [ ] **Step 2: Create the fetcher with singleflight**

Create `internal/api/browse/entities.go`:

```go
package browse

import (
	"context"
	"fmt"

	"github.com/Khan/genqlient/graphql"
	"golang.org/x/sync/singleflight"
	"stash-vr/internal/config"
	"stash-vr/internal/prefix"
	"stash-vr/internal/stash/gql"
	"stash-vr/internal/util"
)

// entitiesGroup deduplicates concurrent fetches of the three entity lists.
var entitiesGroup singleflight.Group

func fetchPerformers(ctx context.Context, client graphql.Client) ([]Entity, error) {
	res, err, _ := entitiesGroup.Do("browse:performers", func() (any, error) {
		// min_scene_count is converted by Stash to GREATER_THAN; passing 1
		// means "scene_count > 0", i.e. only performers with at least one scene.
		resp, err := gql.FindPerformersWithSceneCount(ctx, client, 1, -1)
		if err != nil {
			return nil, fmt.Errorf("FindPerformersWithSceneCount: %w", err)
		}
		out := make([]Entity, 0, len(resp.FindPerformers.Performers))
		for _, p := range resp.FindPerformers.Performers {
			out = append(out, Entity{ID: p.Id, Name: p.Name, SceneCount: p.Scene_count})
		}
		return out, nil
	})
	if err != nil {
		return nil, err
	}
	return res.([]Entity), nil
}

func fetchStudios(ctx context.Context, client graphql.Client) ([]Entity, error) {
	res, err, _ := entitiesGroup.Do("browse:studios", func() (any, error) {
		resp, err := gql.FindAllStudiosWithCount(ctx, client)
		if err != nil {
			return nil, fmt.Errorf("FindAllStudiosWithCount: %w", err)
		}
		out := make([]Entity, 0, len(resp.FindStudios.Studios))
		for _, s := range resp.FindStudios.Studios {
			if s.Scene_count == 0 {
				continue
			}
			out = append(out, Entity{ID: s.Id, Name: s.Name, SceneCount: s.Scene_count})
		}
		return out, nil
	})
	if err != nil {
		return nil, err
	}
	return res.([]Entity), nil
}

func fetchTags(ctx context.Context, client graphql.Client) ([]Entity, error) {
	res, err, _ := entitiesGroup.Do("browse:tags", func() (any, error) {
		// Use existing FindTags with scene_count > 0 filter and sort by scenes_count desc.
		// gql.FindTags variables: tag_filter, sort, direction.
		tagFilter := gql.TagFilterType{
			Scene_count: &gql.IntCriterionInput{
				Value:    0,
				Modifier: gql.CriterionModifierGreaterThan,
			},
		}
		resp, err := gql.FindTags(ctx, client, &tagFilter, util.Ptr("scenes_count"), util.Ptr(gql.SortDirectionEnumDesc))
		if err != nil {
			return nil, fmt.Errorf("FindTags: %w", err)
		}
		excludeSortName := config.Application().ExcludeSortName // typically "hidden"
		out := make([]Entity, 0, len(resp.FindTags.Tags))
		for _, t := range resp.FindTags.Tags {
			if excludeSortName != "" && t.Sort_name == excludeSortName {
				continue
			}
			// Skip ancestor-only tags (those used internally as ancestor markers
			// by decorateTags). Identified by the SvrAncestor prefix.
			if prefix.SvrAncestor != "" && len(t.Sort_name) >= len(prefix.SvrAncestor) && t.Sort_name[:len(prefix.SvrAncestor)] == prefix.SvrAncestor {
				continue
			}
			out = append(out, Entity{ID: t.Id, Name: t.Name, SceneCount: t.Scene_count})
		}
		return out, nil
	})
	if err != nil {
		return nil, err
	}
	return res.([]Entity), nil
}

// LoadSidebar fetches all three entity lists concurrently and returns them
// in a SidebarData struct ready for template rendering.
func LoadSidebar(ctx context.Context, client graphql.Client, activeTab, activeID string) (SidebarData, error) {
	type result struct {
		entities []Entity
		err      error
	}
	performersCh := make(chan result, 1)
	studiosCh := make(chan result, 1)
	tagsCh := make(chan result, 1)

	go func() {
		e, err := fetchPerformers(ctx, client)
		performersCh <- result{e, err}
	}()
	go func() {
		e, err := fetchStudios(ctx, client)
		studiosCh <- result{e, err}
	}()
	go func() {
		e, err := fetchTags(ctx, client)
		tagsCh <- result{e, err}
	}()

	pr, sr, tr := <-performersCh, <-studiosCh, <-tagsCh
	if pr.err != nil {
		return SidebarData{}, pr.err
	}
	if sr.err != nil {
		return SidebarData{}, sr.err
	}
	if tr.err != nil {
		return SidebarData{}, tr.err
	}
	if activeTab == "" {
		activeTab = "perf"
	}
	return SidebarData{
		Performers: pr.entities,
		Studios:    sr.entities,
		Tags:       tr.entities,
		ActiveTab:  activeTab,
		ActiveID:   activeID,
	}, nil
}
```

Notes:
- `config.Application().ExcludeSortName` may not exist as a public field today; check `internal/config/application.go`. If the existing constant is `EXCLUDE_SORT_NAME` exported as `ExcludeSortName`, use it directly. If it's only accessed via a method, use that. The default value is `"hidden"`.
- `prefix.SvrAncestor` is defined in `internal/prefix/constant.go` (referenced in CLAUDE.md). If `Sort_name` is a string (not pointer) we can compare directly per the `# @genqlient(pointer: false)` directive in TagParts.
- `util.Ptr` exists in `internal/util` and is used widely.

  const q = input.value.toLowerCase().trim();

const pageSize = 30

// fetchSceneIDs runs FindSceneIdsByFilter and returns (ids, totalCount).
// scene_filter may be nil (all scenes); filterOpts controls pagination/sort.
func fetchSceneIDs(ctx context.Context, client graphql.Client, sceneFilter *gql.SceneFilterType, page int) (ids []string, total int, err error) {
	perPage := pageSize
	resp, err := gql.FindSceneIdsByFilter(ctx, client, sceneFilter, &gql.FindFilterType{
		Page:      util.Ptr(page),
		Per_page:  util.Ptr(perPage),
		Sort:      util.Ptr("created_at"),
		Direction: util.Ptr(gql.SortDirectionEnumDesc),
	})
	if err != nil {
		return nil, 0, fmt.Errorf("FindSceneIdsByFilter: %w", err)
	}
	out := make([]string, len(resp.FindScenes.Scenes))
	for i, s := range resp.FindScenes.Scenes {
		out[i] = s.Id
	}
	return out, resp.FindScenes.Count, nil
}

// buildCards converts scene IDs into Card structs, fetching scene metadata
// through the library service so the existing cache and decorateTags apply.
func buildCards(ctx context.Context, lib *library.Service, baseURL string, ids []string) ([]Card, error) {
	cards := make([]Card, 0, len(ids))
	for _, id := range ids {
		vd, err := lib.GetScene(ctx, id, false)
		if err != nil {
			log.Ctx(ctx).Warn().Err(err).Str("id", id).Msg("browse: skip scene (fetch error)")
			continue
		}
		if vd == nil || len(vd.SceneParts.Files) == 0 {
			continue
		}
		c := Card{
			ID:           id,
			Title:        vd.Title(),
			Duration:     formatDuration(vd.SceneParts.Files[0].Duration),
			DetailURL:    "/browse/scene/" + url.PathEscape(id),
			DeoVRPlayURL: "/deovr/videoData/" + url.PathEscape(id),
		}
		// Thumbnail: heatmap composite for interactive scenes; screenshot otherwise.
		if vd.SceneParts.Paths.Screenshot != nil {
			if vd.SceneParts.Interactive && vd.SceneParts.Paths.Interactive_heatmap != nil {
				c.ThumbnailURL = heatmap.GetCoverUrl(baseURL, id)
			} else {
				c.ThumbnailURL = stash.ApiKeyed(*vd.SceneParts.Paths.Screenshot)
			}
		}
		// Performers comma-joined.
		names := make([]string, 0, len(vd.SceneParts.Performers))
		for _, p := range vd.SceneParts.Performers {
			names = append(names, p.Name)
		}
		c.Performers = strings.Join(names, ", ")
		if vd.SceneParts.Studio != nil {
			c.Studio = vd.SceneParts.Studio.Name
		}
		cards = append(cards, c)
	}
	return cards, nil
}

func formatDuration(seconds float64) string {
	total := int(seconds)
	h := total / 3600
	m := (total % 3600) / 60
	s := total % 60
	if h > 0 {
		return fmt.Sprintf("%d:%02d:%02d", h, m, s)
	}
	return fmt.Sprintf("%d:%02d", m, s)
}

// pagerURLs returns prev/next URL strings for a base URL with a ?page=N
// param. Returns empty strings on edges. otherParams preserves any
// query params already on the URL (e.g. ?tab=perf).
func pagerURLs(basePath string, page, pageMax int, extraParams url.Values) (prev, next string) {
	mk := func(p int) string {
		v := url.Values{}
		for k, vs := range extraParams {
			for _, x := range vs {
				v.Add(k, x)
			}
		}
		v.Set("page", fmt.Sprintf("%d", p))
		return basePath + "?" + v.Encode()
	}
	if page > 1 {
		prev = mk(page - 1)
	}
	if page < pageMax {
		next = mk(page + 1)
	}
	return
}

// redirectBack issues a 303 to Referer (or /browse/scene/{id} fallback),
// optionally appending an `err` query param when err != "".
func (h *httpHandler) redirectBack(w http.ResponseWriter, r *http.Request, err string) {
	target := r.Header.Get("Referer")
	if target == "" {
		target = "/browse/scene/" + chi.URLParam(r, "id")
	}
	if err != "" {
		sep := "?"
		for _, c := range target {
			if c == '?' {
				sep = "&"
				break
			}
		}
		target += sep + "err=" + url.QueryEscape(err)
	}
	http.Redirect(w, r, target, http.StatusSeeOther)
}
